# tidytv list: Google TV / Android TV streaming sticks (Realme 4K Smart Google TV Stick and similar
# Amlogic/SkyworthDigital sticks running Google TV). Derived from what the community lists in
# github.com/Gathin23/cleanyourtv disable on 84 Google TV devices (MIT). Use `scan` for a plan
# tailored to your exact stick; this file is the conservative, known-safe subset.
#
# One package per line. Lines without '#' are DISABLED for the main user, never uninstalled.
# Apply:  python3 tidytv.py --host <tv ip> apply lists/google-tv-stick.txt
# Undo:   python3 tidytv.py --host <tv ip> restore

# --- Telemetry, feedback, ad plumbing, feeders
com.google.android.feedback                      # Google feedback and crash reporting
com.android.tv.feedbackconsent                   # feedback consent prompts
com.google.android.partnersetup                  # partner setup: OEM/partner customisation pushes
android.autoinstalls.config.google.gtvpai        # auto-installs sponsored apps after setup
com.android.adservices.api                       # Privacy Sandbox ad services
com.android.federatedcompute.services            # federated compute (on-device personalisation)
com.android.ondevicepersonalization.services     # on-device personalisation (ads)
com.google.android.apps.tv.dreamx                # ambient screensaver with promo content
com.google.android.onetimeinitializer            # nothing to do after first boot
com.android.printspooler                         # print spooler on a TV
com.google.android.syncadapters.calendar         # calendar sync
com.google.android.tv.bugreportsender            # bug report uploader
